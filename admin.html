function loadAdminRequests() { // جلب البيانات من المخزن الثابت والمباشر للواجهة دون الحاجة للسيرفر المعطل const requests = JSON.parse(localStorage.getItem('admin_orders')) || []; const container = document.getElementById('requestsContainer'); // تغيير الشارة للون الأخضر لأن الاتصال أصبح محلياً وثابتاً const statusBadge = document.querySelector('.status-badge') || id="liveStatus"; if(statusBadge) { statusBadge.innerText = '🟢 متصل بمستودع البيانات الثابتة'; statusBadge.style.backgroundColor = '#1b5e20'; } container.innerHTML = ''; if (requests.length === 0) { container.innerHTML = '
شاشات المراقبة نظيفة، لا توجد طلبات معلقة حالياً
'; return; } // عرض الطلبات من الأحدث للأقدم [...requests].reverse().forEach(req => { const isWithdraw = req.type && req.type.includes('سحب'); const cardClass = isWithdraw ? 'request-card withdraw-type' : 'request-card'; const typeColor = isWithdraw ? '#ff5252' : '#00e676'; const cardHtml = `

نوع العملية: ${req.type}

المبلغ المطلوب: ${req.amount} USDT

عنوان محفظة العميل: ${req.wallet}

حالة الطلب: ⏳ معلق توقيت الإرسال: ${req.date || 'غير محدد'}
`; container.innerHTML += cardHtml; }); } function clearAllRequests() { if (confirm('هل أنت متأكد من رغبتك في حذف وتصفير جميع الطلبات الحالية من لوحة التحكم؟')) { localStorage.removeItem('admin_orders'); loadAdminRequests(); } } // تحديث فوري مستمر كل ثانيتين setInterval(loadAdminRequests, 2000); window.onload = loadAdminRequests;
